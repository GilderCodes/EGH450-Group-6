# disros definition so we dont need to worrie about ros ip manual setting. 
disros() {
  # Setup for distributed ROS
  export ROS_IP="$(hostname -I | cut -d' ' -f1)"
  echo "Identifying as: $ROS_IP"
 
  if [ "$#" -eq 1 ]
  then
    export ROS_MASTER_URI="http://$1:11311/"
    echo "Connecting to: $ROS_MASTER_URI"
  fi
}

function connect() {
    echo "Launching Connection"
    sleep 2

    # Check if an argument is provided
    if [ -z "$1" ]; then
        echo "No argument provided. Exiting."
        return 1
    fi

    # Store the provided argument
    ARGUMENT=$1

    # Create a new tmux session named 'drone'
    tmux new-session -d -s drone

    tmux set-option -t drone -g mouse on

    # Create three horizontal panes
    tmux split-window -h -t drone:0
    tmux split-window -h -t drone:0

    # Let the terminals populate
    sleep 2

    # Select each pane and run the provided argument followed by XXX
    tmux select-pane -t drone:0.0
    tmux send-keys "disros $ARGUMENT" Enter
    tmux send-keys "roslaunch operator_interfaces customrviz.launch" Enter

    tmux select-pane -t drone:0.1
    tmux send-keys "disros $ARGUMENT" Enter
    tmux send-keys "rqt" Enter

    tmux select-pane -t drone:0.2
    tmux send-keys "disros $ARGUMENT" Enter
    tmux send-keys "rosrun operator_interfaces audio.py" Enter

    # Attach to the session
    tmux attach-session -t drone
}

export -f connect
